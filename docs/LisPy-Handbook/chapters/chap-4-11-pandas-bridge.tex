(defvar *df* (pandas-read-csv "datos.csv"))

                :columns '("ID" "Nombre" "Edad")))

\section{Exploración}

\section{Filtrado y Agrupación}
\begin{lispcode}
(pandas-filter *df* (lambda (row) (> (py:getitem row "Edad") 25)))
(pandas-groupby *df* "Categoria" :aggregations '(("Ventas" . "sum")))
\end{lispcode}
